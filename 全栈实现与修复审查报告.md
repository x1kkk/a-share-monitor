```python
"""Exchange sessions and yesterday's streak cache; no weekday approximation in live mode."""
from __future__ import annotations

import asyncio
import bisect
import json
import logging
import os
import sys
from dataclasses import dataclass, field
from datetime import date, datetime

from config import SHANGHAI, Settings

log = logging.getLogger(__name__)

# A subprocess bounds AKShare's synchronous network work and can actually be killed
# on timeout (cancelling asyncio.to_thread would leave its requests running).
AK_BOOTSTRAP = r'''
import contextlib, datetime, json, sys
with contextlib.redirect_stdout(sys.stderr):
    import akshare as ak
    dates = sorted(str(x)[:10] for x in ak.tool_trade_date_hist_sina()["trade_date"])
    today = sys.argv[1]
    previous = [x for x in dates if x < today]
    streaks = {}
    streaks_ready = False
    if previous:
        try:
            pool = ak.stock_zt_pool_em(date=previous[-1].replace("-", ""))
            for _, row in pool.iterrows():
                streaks[str(row["代码"]).zfill(6)] = int(row["连板数"])
            streaks_ready = True
        except Exception:
            pass
print(json.dumps({"sessions": dates, "streaks": streaks, "streaks_ready": streaks_ready}))
'''


@dataclass
class MarketCalendar:
    sessions: list[date] = field(default_factory=list)
    yesterday_streaks: dict[str, int] = field(default_factory=dict)
    streaks_ready: bool = False
    loaded_for: date | None = None
    error: str | None = "calendar not initialized"

    async def load(self, cfg: Settings, today: date) -> None:
        process = None
        if self.loaded_for != today:
            self.yesterday_streaks.clear()
            self.streaks_ready = False
        try:
            process_env = {key: value for key, value in os.environ.items()
                           if key.upper() not in {"HTTP_PROXY", "HTTPS_PROXY", "ALL_PROXY", "NO_PROXY"}}
            if cfg.http_proxy:
                process_env.update(HTTP_PROXY=cfg.http_proxy, HTTPS_PROXY=cfg.http_proxy)
            process = await asyncio.create_subprocess_exec(
                sys.executable, "-c", AK_BOOTSTRAP, today.isoformat(),
                stdout=asyncio.subprocess.PIPE, stderr=asyncio.subprocess.DEVNULL,
                env=process_env,
            )
            stdout, _ = await asyncio.wait_for(process.communicate(), cfg.calendar_timeout)
            if process.returncode:
                raise ValueError("AKShare calendar process failed")
            data = json.loads(stdout)
            sessions = sorted({date.fromisoformat(x) for x in data["sessions"]})
            if not sessions or sessions[-1] < today:
                raise ValueError("calendar does not cover today; update AKShare")
            self.sessions = sessions
            self.yesterday_streaks = {k: int(v) for k, v in data["streaks"].items()}
            self.streaks_ready = bool(data["streaks_ready"])
            self.loaded_for, self.error = today, None
            log.info("盘前缓存：交易日=%d 昨日连板记录=%d", len(sessions), len(self.yesterday_streaks))
        except asyncio.CancelledError:
            raise
        except Exception as exc:
            self.loaded_for = today
            self.error = str(exc)
            self.yesterday_streaks.clear()
            self.streaks_ready = False
            log.warning("盘前初始化不可用：%s；实盘严格过滤关闭，模拟模式仍可运行", self.error)
        finally:
            if process is not None and process.returncode is None:
                process.kill()
                await process.wait()

    def covered(self, today: date) -> bool:
        return bool(self.error is None and self.sessions and self.sessions[-1] >= today and self.loaded_for == today)

    def listing_is_mature(self, listed: date | None, today: date, minimum: int = 15) -> bool:
        if listed is None or listed > today or not self.covered(today):
            return False
        # At the 15th session (listing day inclusive) the stock is eligible.
        return bisect.bisect_right(self.sessions, today) - bisect.bisect_left(self.sessions, listed) >= minimum

    def session(self, now: datetime) -> str:
        now = now.astimezone(SHANGHAI)
        if not self.covered(now.date()):
            return "calendar_unavailable"
        if now.date() not in self.sessions:
            return "closed"
        minute = now.hour * 60 + now.minute
        if 555 <= minute < 690:
            return "morning"
        if 780 <= minute < 900:
            return "afternoon"
        return "closed"
```

```python
def number(value: object, default: float = 0.0) -> float:
    try:
        result = float(value)
        return result if math.isfinite(result) else default
    except (ValueError, TypeError):
        return default


def limit_price(pre_close: float) -> float:
    return float((Decimal(str(pre_close)) * Decimal("1.10")).quantize(Decimal("0.01"), rounding=ROUND_HALF_UP))


@dataclass(frozen=True)
class Quote:

def parse_quote(row: dict, *, now: datetime | None = None) -> Quote | None:
    code, name = str(row.get("f12", "")), str(row.get("f14", "")).strip()
    if not re.fullmatch(r"(?:60|00)\d{4}", code) or not name or "ST" in name.upper():
        return None
    snapshot_time = (now or datetime.now(SHANGHAI)).astimezone(SHANGHAI)
    minute = snapshot_time.hour * 60 + snapshot_time.minute
    # During auction Eastmoney uses '-' (not numeric zero) before the first
    # trade. Only normalize absent counters for genuinely untraded quotes;
    # missing flows on a traded quote remain invalid below.
    missing = (None, "", "-")
    if 555 <= minute < 570 and (row.get("f5") in missing or number(row.get("f5"), -1) == 0):
        row = dict(row)
        for key in ("f5", "f6", "f8", "f62", "f66", "f72"):
            if row.get(key) in missing:
                row[key] = 0
    price, volume, previous = number(row.get("f2")), number(row.get("f5"), -1), number(row.get("f18"))
    # Indicative auction prices are valid before the first trade (volume == 0).
    if price <= 0 or volume < 0 or previous <= 0:
        return None
    # A missing cash-flow field must not turn yesterday's negative value into fake inflow.
    if any(not math.isfinite(number(row.get(key), math.nan))
           for key in ("f3", "f6", "f8", "f21", "f62", "f66", "f72")):
        return None
    try:
        listed = datetime.strptime(str(row.get("f26", "")), "%Y%m%d").date()
    except ValueError:
        listed = None
    return Quote(
        code, name, price, number(row.get("f3")), volume, number(row.get("f6")),
        number(row.get("f8")), previous, number(row.get("f21")), listed,
        number(row.get("f62")), number(row.get("f66")), number(row.get("f72")),
        str(row.get("f100") or ""), str(row.get("f103") or ""),
        number(row.get("f15")), number(row.get("f124")) or None,
    )


def parse_books(text: str) -> dict[str, Book]:
    result = {}
    for code, body in re.findall(r'v_(?:sh|sz)(\d{6})="([^"]*)"', text):
        fields = body.split("~")
        if len(fields) <= 30:
            continue
        try:
            ts = datetime.strptime(fields[30], "%Y%m%d%H%M%S").replace(tzinfo=SHANGHAI).timestamp()
            price, buy, sell = (number(fields[i], -1) for i in (3, 10, 20))
            if price > 0 and buy >= 0 and sell >= 0:
                result[code] = Book(price, buy, sell, ts)
        except ValueError:
            continue
    return result


class Fetcher:
    def __init__(self, client: httpx.AsyncClient, cfg: Settings):
        self.client, self.cfg = client, cfg
        self.semaphore = asyncio.Semaphore(cfg.max_concurrency)
        self.last_raw: list[Quote] = []
        self.last_total: int | None = None

    async def _page(self, page: int, request_timeout: float | None = None) -> tuple[int, list[dict]]:
        params = {"pn": page, "pz": self.cfg.page_size, "po": 0, "np": 1,
                  "ut": "bd1d9ddb04089700cf9c27f6f7426281",
                  "fltt": 2, "invt": 2, "fid": "f12", "fs": self.cfg.market_filter,
                  "fields": self.cfg.fields}
        async with self.semaphore:
            response = await self.client.get(self.cfg.eastmoney_url, params=params,
                                             timeout=request_timeout or self.cfg.http_timeout)
            response.raise_for_status()
        payload = response.json()
        data = payload.get("data") if isinstance(payload, dict) else None
        if not isinstance(data, dict) or not isinstance(data.get("diff"), list):
            raise ValueError("Eastmoney response has no diff list")
        if any(not isinstance(row, dict) for row in data["diff"]):
            raise ValueError("invalid quote row")
        try:
            total = int(data.get("total", 0))
        except (ValueError, TypeError) as exc:
            raise ValueError("invalid market total") from exc
        return total, data["diff"]

    async def fetch_market(self, *, bootstrap: bool = False) -> list[Quote]:
        # Reject partial/duplicate pages; never feed a partial market into percentiles.
        deadline = self.cfg.bootstrap_deadline if bootstrap else self.cfg.eastmoney_deadline
        request_timeout = self.cfg.bootstrap_http_timeout if bootstrap else self.cfg.http_timeout
        async with asyncio.timeout(deadline):
            if self.last_total is None:
                total, first = await self._page(1, request_timeout)
                pages = [(total, first)]
                start_page = 2
            else:
                total, pages, start_page = self.last_total, [], 1
            if total <= 0:
                raise ValueError("empty market snapshot")
            page_count = math.ceil(total / self.cfg.page_size)
            if page_count > self.cfg.max_pages:
                raise ValueError("unexpected market size")
            pages += await asyncio.gather(*(self._page(p, request_timeout) for p in range(start_page, page_count + 1)),
                                          return_exceptions=True)
            rows = []
            for page in pages:
                if isinstance(page, BaseException):
                    raise page
                other_total, batch = page
                if other_total != total:
                    self.last_total = None
                    raise ValueError("market universe changed during pagination")
                rows.extend(batch)
            codes = [str(row.get("f12", "")) for row in rows]
            if len(rows) != total or len(set(codes)) != total:
                raise ValueError("incomplete or duplicated market pages")
            snapshot_time = datetime.now(SHANGHAI)
            quotes = [q for row in rows if (q := parse_quote(row, now=snapshot_time)) is not None]
            if not quotes:
                raise ValueError("no tradable main-board quotes")
            self.last_raw = quotes
            self.last_total = total
            return quotes

    async def fetch_books(self, codes: set[str], now: datetime) -> dict[str, Book]:
        if not codes:
            return {}

        async def batch(items: list[str]) -> dict[str, Book]:

class Simulator:
    def __init__(self, seeds: list[Quote], cfg: Settings, source: str):
        self.rng = random.Random(cfg.mock_seed)
        self.source = source
        self.base = {q.code: q for q in seeds}
        self.current = dict(self.base)
        self.tick = 0
        codes = list(self.base)
        self.cohort = self.rng.sample(codes, min(cfg.mock_cohort, len(codes)))
        self.sealed = set(self.cohort[:max(1, len(self.cohort) // 2)])

    def step(self, now: datetime) -> tuple[list[Quote], dict[str, Book]]:
        self.tick += 1
        # First tick is the baseline. Then a random subset rises by 0.8%-2.5%.
        phase = self.tick % 240
        stimulated = self.rng.sample(self.cohort, min(3, len(self.cohort))) if 2 <= phase <= 14 else []
        books = {}
        for code, base in self.base.items():
            old = self.current[code]
            ceiling = limit_price(base.pre_close)
            price, flow = old.price, old.main_inflow
            if code in self.cohort and (self.tick == 1 or phase == 0):
                price = round(base.pre_close * 0.99, 2)
                flow = 0.0
            elif code in stimulated:
```

```python
def percentiles(values: dict[str, float]) -> dict[str, float]:
    """Average ranks for ties. A flat market gets 50, not a spurious 100."""
    ordered = sorted(values, key=values.get)
    result = {}
    start, count = 0, len(ordered)
    while start < count:
        end = start + 1
        while end < count and values[ordered[end]] == values[ordered[start]]:
            end += 1
        score = 50.0 if count == 1 else ((start + end - 1) / 2) / (count - 1) * 100
        for code in ordered[start:end]:
            result[code] = score
        start = end
    return result
```

```python
"""One cooldown gate shared by browser events and a bounded WeCom queue."""
from __future__ import annotations

import asyncio
import logging
import time
import uuid
from urllib.parse import urlparse

import httpx

from config import Settings

log = logging.getLogger(__name__)
EVENT_NAMES = {"anomaly": "实时异动", "limit_up": "首次封板", "break_open": "炸板告警"}


class Notifier:
    def __init__(self, client: httpx.AsyncClient, cfg: Settings):
        self.client, self.cfg = client, cfg
        self.last_notify_time: dict[tuple[str, str], float] = {}
        self.queue: asyncio.Queue[dict] = asyncio.Queue(maxsize=1000)
        self.sent = self.failed = self.suppressed = self.dropped = 0
        if cfg.wechat_webhook_url:
            url = urlparse(cfg.wechat_webhook_url)
            if url.scheme != "https" or url.hostname != "qyapi.weixin.qq.com" or url.path != "/cgi-bin/webhook/send":
                raise ValueError("WECHAT_WEBHOOK_URL must be an HTTPS WeCom group robot webhook")

    def publish(self, events: list[dict], *, clock: float | None = None) -> list[dict]:
        clock = time.monotonic() if clock is None else clock
        accepted = []
        # Prune old cooldown keys, retaining a bounded daily dictionary.
        self.last_notify_time = {key: ts for key, ts in self.last_notify_time.items()
                                 if clock - ts < self.cfg.notify_cooldown}
        for raw in events:
            # Simulation never consumes the live-event cooldown budget.
            code = ("mock:" if raw["simulated"] else "live:") + raw["code"]
            key = (code, raw["event_type"])
            if key in self.last_notify_time:
                self.suppressed += 1
                log.info("冷却抑制 %s %s %s", raw["code"], raw["name"], EVENT_NAMES[raw["event_type"]])
                continue
            self.last_notify_time[key] = clock
            event = {**raw, "id": uuid.uuid4().hex}
            accepted.append(event)
            log.info("%s[%s] %s %s 价格=%.2f 涨幅=%.2f%% 涨速=%s 分数=%s",
                     "模拟" if raw["simulated"] else "实盘", EVENT_NAMES[raw["event_type"]],
                     raw["code"], raw["name"], raw["price"], raw["pct"],
                     raw.get("speed_2m", "-"), raw.get("score", "-"))
            if self.cfg.wechat_webhook_url and (not raw["simulated"] or self.cfg.mock_wechat):
                try:
                    self.queue.put_nowait(event)
                except asyncio.QueueFull:
                    self.dropped += 1
                    log.error("企业微信队列已满：事件 %s 未投递", event["id"])
        return accepted

    @staticmethod
    def markdown(event: dict) -> str:
        name = str(event["name"]).replace("<", "").replace(">", "").replace("\n", " ")
        mode = "模拟演练" if event["simulated"] else "实盘监控"
        content = (f"### 【{mode}】{EVENT_NAMES[event['event_type']]}\n"
                   f"**{name} ({event['code']})**\n"
                   f"> 价格：{event['price']:.2f}　涨幅：{event['pct']:.2f}%\n"
                   f"> 时间：{event['time']}\n")
        if "speed_2m" in event:
            content += f"> 两分钟涨速：{event['speed_2m']:.2f}%　强度：{event['score']:.2f}\n"
        if "break_count" in event:
            content += f"> 今日开板次数：{event['break_count']}\n"
        return content + f"\n事件编号：{event['id']}"

    async def worker(self) -> None:
        while True:
            event = await self.queue.get()
            try:
                # This worker is independent of the two-second polling task.
                response = await self.client.post(
                    self.cfg.wechat_webhook_url,
                    json={"msgtype": "markdown", "markdown": {"content": self.markdown(event)}},
                    timeout=5.0,
                )
                response.raise_for_status()
                payload = response.json()
                if not isinstance(payload, dict) or payload.get("errcode") != 0:
                    raise ValueError("WeCom rejected robot message")
                self.sent += 1
            except (httpx.HTTPError, ValueError):
                # No ambiguous retries: a timed-out POST may already have been delivered.
                self.failed += 1
                log.error("企业微信发送失败 event=%s；检查网络、Webhook 和机器人限流", event["id"])
            finally:
                self.queue.task_done()
            await asyncio.sleep(3.1)

    def status(self) -> dict:
        return {"configured": bool(self.cfg.wechat_webhook_url), "mock_enabled": self.cfg.mock_wechat,
                "queued": self.queue.qsize(), "sent": self.sent, "failed": self.failed,
                "suppressed": self.suppressed, "dropped": self.dropped}
```
